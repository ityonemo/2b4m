## Exercises

**Exercise 19.** Let $f: A \to B$ and $g: B \to C$ be invertible mappings. Show
that $(g \circ f)^{-1} = f^{-1} \circ g^{-1}$.

Two supporting facts. First, the **identity laws**
$\operatorname{id} \circ f = f$ and $f \circ \operatorname{id} = f$. Second,
**uniqueness of the inverse**: a left inverse equals a right inverse, since if
$h \circ k = \operatorname{id}$ and $k \circ m = \operatorname{id}$ then
$m = \operatorname{id}\circ m = (h\circ k)\circ m = h\circ(k\circ m) =
h\circ\operatorname{id} = h$.

```2b4m
theorem composeIdentityLeft: forall f: Fn; compose(id, f) = f
proof
  @conclusion |
    forall f: Fn; compose(id, f) = f
    [using extensionality_quantified(funcExt) composeApply identityApply]
qed

theorem composeIdentityRight: forall f: Fn; compose(f, id) = f
proof
  @conclusion |
    forall f: Fn; compose(f, id) = f
    [using extensionality_quantified(funcExt) composeApply identityApply]
qed

// uniqueness: h∘k = id and k∘m = id force m = h.
theorem inverseUnique: forall h, k, m: Fn;
  compose(h, k) = id -> compose(k, m) = id -> m = h
proof
  @generalize-h |
    fix h: Fn {
      @generalize-k |
        fix k: Fn {
          @generalize-m |
            fix m: Fn {
              @given-left |
                assume compose(h, k) = id {
                  @given-right |
                    assume compose(k, m) = id {
                      @left |
                        compose(h, k) = id
                        [by hypothesis given-left]
                      @right |
                        compose(k, m) = id
                        [by hypothesis given-right]
                      @m-is-id-m |
                        compose(id, m) = m
                        [using specialize composeIdentityLeft(m)]
                      @start |
                        m = compose(id, m)
                        [by symmetry m-is-id-m]
                      @insert-hk |
                        m = compose(compose(h, k), m)
                        [by rewrite left start]
                      @regroup |
                        compose(compose(h, k), m) = compose(h, compose(k, m))
                        [using specialize composeAssoc(h, k, m)]
                      @regrouped |
                        m = compose(h, compose(k, m))
                        [by rewrite regroup insert-hk]
                      @collapse-right |
                        m = compose(h, id)
                        [by rewrite right regrouped]
                      @h-id-is-h |
                        compose(h, id) = h
                        [using specialize composeIdentityRight(h)]
                      @conclusion-m-is-h |
                        m = h
                        [by rewrite h-id-is-h collapse-right]
                    }
                  @discharge-right |
                    compose(k, m) = id -> m = h
                    [by discharge given-right]
                }
              @discharge-left |
                compose(h, k) = id -> compose(k, m) = id -> m = h
                [by discharge given-left]
            }
          @discharge-m |
            forall m: Fn; compose(h, k) = id -> compose(k, m) = id -> m = h
            [by generalize generalize-m]
        }
      @discharge-k |
        forall k, m: Fn; compose(h, k) = id -> compose(k, m) = id -> m = h
        [by generalize generalize-k]
    }
  @conclusion |
    forall h, k, m: Fn; compose(h, k) = id -> compose(k, m) = id -> m = h
    [by generalize generalize-h]
qed

// a composite of invertibles is invertible, with witness f⁻¹∘g⁻¹.
theorem composeInvertible: forall g, f: Fn;
  invertible(g) -> invertible(f) -> invertible(compose(g, f))
proof
  @generalize-g |
    fix g: Fn {
      @generalize-f |
        fix f: Fn {
          @given-g-invertible |
            assume invertible(g) {
              @given-f-invertible |
                assume invertible(f) {
                  @g-invertible |
                    invertible(g)
                    [by hypothesis given-g-invertible]
                  @f-invertible |
                    invertible(f)
                    [by hypothesis given-f-invertible]
                  @f-cancels-on-the-right |
                    compose(f, inverse(f)) = id
                    [using specialize inverseRight(f) f-invertible]
                  @f-cancels-on-the-left |
                    compose(inverse(f), f) = id
                    [using specialize inverseLeft(f) f-invertible]
                  @g-cancels-on-the-right |
                    compose(g, inverse(g)) = id
                    [using specialize inverseRight(g) g-invertible]
                  @g-cancels-on-the-left |
                    compose(inverse(g), g) = id
                    [using specialize inverseLeft(g) g-invertible]
                  // (g∘f)∘(f⁻¹∘g⁻¹) = g∘(f∘(f⁻¹∘g⁻¹)) = g∘((f∘f⁻¹)∘g⁻¹)
                  //   = g∘(id∘g⁻¹) = g∘g⁻¹ = id
                  @right-reassociate-outer |
                    compose(compose(g, f), compose(inverse(f), inverse(g)))
                      = compose(g, compose(f, compose(inverse(f), inverse(g))))
                    [using specialize composeAssoc(g, f, compose(inverse(f), inverse(g)))]
                  @f-then-inverses-reassociate |
                    compose(compose(f, inverse(f)), inverse(g)) = compose(f, compose(inverse(f), inverse(g)))
                    [using specialize composeAssoc(f, inverse(f), inverse(g))]
                  @identity-then-inverse-g |
                    compose(id, inverse(g)) = inverse(g)
                    [using specialize composeIdentityLeft(inverse(g))]
                  @right-composite-is-identity |
                    compose(compose(g, f), compose(inverse(f), inverse(g))) = id
                    [using chain right-reassociate-outer f-then-inverses-reassociate f-cancels-on-the-right identity-then-inverse-g g-cancels-on-the-right]
                  // (f⁻¹∘g⁻¹)∘(g∘f) = f⁻¹∘(g⁻¹∘(g∘f)) = f⁻¹∘((g⁻¹∘g)∘f)
                  //   = f⁻¹∘(id∘f) = f⁻¹∘f = id
                  @left-reassociate-outer |
                    compose(compose(inverse(f), inverse(g)), compose(g, f))
                      = compose(inverse(f), compose(inverse(g), compose(g, f)))
                    [using specialize composeAssoc(inverse(f), inverse(g), compose(g, f))]
                  @g-inverse-then-g-reassociate |
                    compose(compose(inverse(g), g), f) = compose(inverse(g), compose(g, f))
                    [using specialize composeAssoc(inverse(g), g, f)]
                  @identity-then-f |
                    compose(id, f) = f
                    [using specialize composeIdentityLeft(f)]
                  @left-composite-is-identity |
                    compose(compose(inverse(f), inverse(g)), compose(g, f)) = id
                    [using chain left-reassociate-outer g-inverse-then-g-reassociate g-cancels-on-the-left identity-then-f f-cancels-on-the-left]
                  @two-sided |
                    compose(compose(inverse(f), inverse(g)), compose(g, f)) = id
                      and compose(compose(g, f), compose(inverse(f), inverse(g))) = id
                    [by both left-composite-is-identity right-composite-is-identity]
                  @inverse-exists |
                    exists h: Fn; compose(h, compose(g, f)) = id and compose(compose(g, f), h) = id
                    [by witness(compose(inverse(f), inverse(g))) two-sided]
                  @invertible-def-at-composite |
                    invertible(compose(g, f)) iff (exists h: Fn; compose(h, compose(g, f)) = id and compose(compose(g, f), h) = id)
                    [using specialize invertibleDef(compose(g, f))]
                  @two-sided-inverse-makes-composite-invertible |
                    (exists h: Fn; compose(h, compose(g, f)) = id and compose(compose(g, f), h) = id) -> invertible(compose(g, f))
                    [by equiv_converse invertible-def-at-composite]
                  @conclusion-composite-invertible |
                    invertible(compose(g, f))
                    [by modus_ponens two-sided-inverse-makes-composite-invertible inverse-exists]
                }
              @f-invertible-implies-composite-invertible |
                invertible(f) -> invertible(compose(g, f))
                [by discharge given-f-invertible]
            }
          @g-invertible-implies-composite-invertible |
            invertible(g) -> invertible(f) -> invertible(compose(g, f))
            [by discharge given-g-invertible]
        }
      @discharge-f |
        forall f: Fn; invertible(g) -> invertible(f) -> invertible(compose(g, f))
        [by generalize generalize-f]
    }
  @conclusion |
    forall g, f: Fn; invertible(g) -> invertible(f) -> invertible(compose(g, f))
    [by generalize generalize-g]
qed

// (g∘f)⁻¹ = f⁻¹∘g⁻¹.  f⁻¹∘g⁻¹ is a right inverse of g∘f (p2, by regrouping);
// inverse(g∘f) is a left inverse of it (inverseLeft); left = right (inverseUnique).
theorem inverseOfComposite: forall g, f: Fn;
  invertible(g) -> invertible(f)
  -> inverse(compose(g, f)) = compose(inverse(f), inverse(g))
proof
  @generalize-g |
    fix g: Fn {
      @generalize-f |
        fix f: Fn {
          @given-g-invertible |
            assume invertible(g) {
              @given-f-invertible |
                assume invertible(f) {
                  @g-invertible |
                    invertible(g)
                    [by hypothesis given-g-invertible]
                  @f-invertible |
                    invertible(f)
                    [by hypothesis given-f-invertible]
                  @composite-invertible |
                    invertible(compose(g, f))
                    [using specialize composeInvertible(g, f) g-invertible f-invertible]
                  @assoc |
                    forall p, q, r: Fn; compose(compose(p, q), r) = compose(p, compose(q, r))
                    [by cite composeAssoc]
                  @f-cancels-on-the-right |
                    compose(f, inverse(f)) = id
                    [using specialize inverseRight(f) f-invertible]
                  @g-cancels-on-the-right |
                    compose(g, inverse(g)) = id
                    [using specialize inverseRight(g) g-invertible]
                  @left-inverse-of-composite |
                    compose(inverse(compose(g, f)), compose(g, f)) = id
                    [using specialize inverseLeft(compose(g, f)) composite-invertible]
                  // (g∘f)∘(f⁻¹∘g⁻¹) = g∘(f∘(f⁻¹∘g⁻¹)) = g∘((f∘f⁻¹)∘g⁻¹)
                  //                 = g∘(id∘g⁻¹) = g∘g⁻¹ = id
                  @regroup-outer |
                    compose(compose(g, f), compose(inverse(f), inverse(g))) = compose(g, compose(f, compose(inverse(f), inverse(g))))
                    [by apply_at(g, f, compose(inverse(f), inverse(g))) assoc]
                  @regroup-inner |
                    compose(compose(f, inverse(f)), inverse(g)) = compose(f, compose(inverse(f), inverse(g)))
                    [by apply_at(f, inverse(f), inverse(g)) assoc]
                  @regrouped |
                    compose(compose(g, f), compose(inverse(f), inverse(g))) = compose(g, compose(compose(f, inverse(f)), inverse(g)))
                    [by rewrite regroup-inner regroup-outer]
                  @f-cancelled |
                    compose(compose(g, f), compose(inverse(f), inverse(g))) = compose(g, compose(id, inverse(g)))
                    [by rewrite f-cancels-on-the-right regrouped]
                  @identity-then-inverse-g |
                    compose(id, inverse(g)) = inverse(g)
                    [using specialize composeIdentityLeft(inverse(g))]
                  @identity-collapsed |
                    compose(compose(g, f), compose(inverse(f), inverse(g))) = compose(g, inverse(g))
                    [by rewrite identity-then-inverse-g f-cancelled]
                  @right-inverse-of-composite |
                    compose(compose(g, f), compose(inverse(f), inverse(g))) = id
                    [by rewrite g-cancels-on-the-right identity-collapsed]
                  @inverses-swapped |
                    compose(inverse(f), inverse(g)) = inverse(compose(g, f))
                    [using specialize inverseUnique(inverse(compose(g, f)), compose(g, f), compose(inverse(f), inverse(g))) left-inverse-of-composite right-inverse-of-composite]
                  @conclusion-inverse-of-composite |
                    inverse(compose(g, f)) = compose(inverse(f), inverse(g))
                    [by symmetry inverses-swapped]
                }
              @f-invertible-implies-inverse-of-composite |
                invertible(f) -> inverse(compose(g, f)) = compose(inverse(f), inverse(g))
                [by discharge given-f-invertible]
            }
          @g-invertible-implies-inverse-of-composite |
            invertible(g) -> invertible(f) -> inverse(compose(g, f)) = compose(inverse(f), inverse(g))
            [by discharge given-g-invertible]
        }
      @discharge-f |
        forall f: Fn; invertible(g) -> invertible(f) -> inverse(compose(g, f)) = compose(inverse(f), inverse(g))
        [by generalize generalize-f]
    }
  @conclusion |
    forall g, f: Fn; invertible(g) -> invertible(f) -> inverse(compose(g, f)) = compose(inverse(f), inverse(g))
    [by generalize generalize-g]
qed
```
